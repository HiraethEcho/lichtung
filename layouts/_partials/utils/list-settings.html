{{- $page := .page -}} {{- $pages := .pages -}} {{- $limit := default true .limit -}} {{- if $limit -}} {{- with default $page.Site.Params.ListLimit $page.Params.ListLimit -}} {{- $pages = first . $pages -}} {{- end -}} {{- end -}} {{- $liststyle := default .defaultlist (or $page.Params.ListStyle $page.Site.Params.ListStyle) -}} {{- $itemstyle := default .defaultitem (default $page.Site.Params.ItemStyle $page.Params.ItemStyle) -}} {{- $visiblepages := $pages -}} {{- if eq $liststyle "date" -}} {{- /* Match the date list's exclusion of posts without a publish date. */ -}} {{- $visiblepages = where $pages "PublishDate.Year" "ne" 1 -}} {{- end -}} {{- $show := default true (default $page.Site.Params.ShowList $page.Params.ShowList) -}} {{- $show = and $show (gt (len $visiblepages) 0) -}} {{- return (dict "pages" $pages "liststyle" $liststyle "itemstyle" $itemstyle "show" $show) -}}
